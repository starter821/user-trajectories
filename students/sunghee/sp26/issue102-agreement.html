# Considering each post that ever had a note attached
attached = notes.dropna(subset=["noteFirstCrhTime"]).copy()

# find the time that a note was first attached
first_attach = (
    attached.groupby("tweetId")["noteFirstCrhTime"]
    .min()
    .reset_index(name="firstAttachTime")
)

# merge back
df = notes.merge(first_attach, on="tweetId", how="inner")

# notes written BEFORE first attachment
before_df = df[df["noteTime"] < df["firstAttachTime"]].copy()
